7class AccumulatorMixin: 

8 """Mixin providing lazy first-batch accumulation for torchmetrics states. 

9 

10 Several metrics in this package hold a running-sum state that starts as an

61def binary_ice_edge( 

62 ice_mask: torch.Tensor, land_mask: torch.Tensor | None = None 

63) -> torch.Tensor: 

64 """Boolean ice-edge map: True for ice cells that border a non-ice ocean cell. 

65 

66 Cells beyond the grid boundary are treated as matching the cell they border (via 

67 replicate padding), rather than being manufactured as non-ice: the domain's own 

68 edge is not itself an ice/ocean transition, so it should not be able to invent a 

69 disagreement just because a real edge would need one more ring of pixels to 

70 resolve. A genuine ice edge that runs along the domain boundary is still detected 

71 normally, since it disagrees with its interior (in-grid) neighbors regardless. 

72 

73 Args: 

74 ice_mask: Boolean tensor of shape (N, H, W). 

75 land_mask: Boolean tensor of shape (H, W), True for ocean cells and False for 

76 land. When given, land cells are excluded from the edge test — both as 

77 neighbors (an ice cell bordering only land is not counted as an edge cell, 

78 though it is still counted if it also borders true open water) and from the 

79 returned map itself, so a land cell is never reported as an edge cell even 

80 if its raw (unmasked) value happens to read as "ice". 

81 

82 Returns: 

83 Boolean tensor of shape (N, H, W), True for ice-edge cells. 

84 

85 """ 

86 comparison_mask = ice_mask 

87 if land_mask is not None: 

88 comparison_mask = ice_mask | ~land_mask.bool() 

89 padded = F.pad(comparison_mask.float(), (1, 1, 1, 1), mode="replicate").bool() 

90 up = padded[:, :-2, 1:-1] 

91 down = padded[:, 2:, 1:-1] 

92 left = padded[:, 1:-1, :-2] 

93 right = padded[:, 1:-1, 2:] 

94 edge = ice_mask & ( 

95 (ice_mask != up) | (ice_mask != down) | (ice_mask != left) | (ice_mask != right) 

96 ) 

97 if land_mask is not None: 

98 edge = edge & land_mask.bool() 

99 return edge
