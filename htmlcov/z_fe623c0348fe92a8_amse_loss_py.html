1"""Adjusted-MSE (AMSE) spectral anti-blur loss on a flat grid. 

2 

3Adaptation of the spherical-harmonic AMSE loss of Subich et al. (2025), 

4"Fixing the Double Penalty in Data-Driven Weather Forecasting Through a 

5Modified Spherical Harmonic Loss Function" (arXiv:2501.19374), to the flat 

6equal-area EASE2 grids used by icenet-mp. Independent reimplementation of the 

7published formula; reference implementation (spherical, JAX) at 

8https://github.com/csubich/graphcast (branch ``amse``). 

9 

10Pointwise squared-error losses reward damping the amplitude of scales the 

11model cannot predict exactly (the "double penalty"): per scale group, the 

12MSE-optimal amplitude ratio equals the coherence with the target, so partially 

13predictable fine scales are systematically blurred. AMSE removes this 

14incentive. Writing per scale group k the powers P_k(x), P_k(y) and coherence 

15Coh_k(x, y) of prediction x and target y, 

16 

17 MSE(x, y) = sum_k [ (sqrt(P_k(x)) - sqrt(P_k(y)))^2 

18 + 2 * sqrt(P_k(x) * P_k(y)) * (1 - Coh_k(x, y)) ] 

19 AMSE(x, y) = sum_k [ (sqrt(P_k(x)) - sqrt(P_k(y)))^2 

20 + 2 * max(P_k(x), P_k(y)) * (1 - Coh_k(x, y)) ] 

21 

22so under AMSE the decorrelation penalty can no longer be reduced by shrinking 

23the predicted amplitude, and the amplitude optimum at any fixed coherence is 

24P_k(x) = P_k(y): the target's power spectrum is preserved scale by scale. 

25 

26On the flat grid the orthonormal decomposition is the 2D DFT and the scale 

27groups are annular bins of the isotropic wavenumber |k| = sqrt(kx^2 + ky^2); 

28Parseval's identity makes the group decomposition of MSE exact, which is the 

29only property the construction needs (Subich et al., Sec. 2.2 and 4.2). 

30 

31Scope: designed and tested for spatial-field predictions (single-stage 

32encode-process-decode). Using this loss with models whose training loss lives 

33in a different space (e.g. DDPM v-space) is untested. 

34"""

65class AMSELoss(nn.Module): 

66 """Adjusted-MSE (anti-double-penalty) spectral loss on a flat grid. 

67 

68 Two modes: 

69 

70 - ``"hybrid"`` (default): the repo's standard Huber loss plus 

71 ``spectral_weight`` times the AMSE *excess* 

72 ``sum_k 2 * (max(P_k(x), P_k(y)) - sqrt(P_k(x) * P_k(y))) * (1 - Coh_k)``, 

73 which is >= 0, zero iff per-bin spectra match or coherence is perfect, 

74 and reduces to the unmodified Huber control at ``spectral_weight=0``. 

75 - ``"pure"``: the AMSE formula itself (per-bin amplitude + decorrelation 

76 terms plus a direct DC term), averaged over fields. 

77 

78 Both modes sum the annular bins UNWEIGHTED by default, as in the Subich 

79 reference. ``wavenumber_weight="fastnet"`` instead upweights each bin by 

80 ``gamma_k = max(N_k * k**sqrt(3), 1)`` (FastNet, arXiv:2509.17601), which 

81 moves the penalty from the large scales that dominate a steep sea-ice 

82 spectrum onto the fine scales where the ice edge lives; the total is held 

83 fixed so ``spectral_weight`` stays comparable across the two settings (see 

84 ``_weight_bins``). Default OFF: unset, the loss is bit-for-bit unchanged. 

85 

86 Numerical guards follow the reference implementation: ``eps`` inside the 

87 geometric-mean square root (the coherence denominator is singular for a 

88 zero field), coherence clamped at 1 from above (Cauchy-Schwarz can be 

89 violated by the eps), the mean/DC component handled as a direct squared 

90 difference (catastrophic cancellation), and all spectral arithmetic in 

91 float32. 

92 """ 

93 

94 def __init__( # noqa: PLR0913 - flat config-driven knobs, all defaulted 

95 self, 

96 *, 

97 mode: AMSEMode = "hybrid", 

98 spectral_weight: float = 0.1, 

99 delta: float = 0.5, 

100 merge_bins_below: int = 4, 

101 eps: float = 1e-12, 

102 static_ref_path: str | None = None, 

103 wavenumber_weight: str | None = None, 

104 ) -> None: 

105 """Initialise the AMSE loss. 

106 

107 Args: 

108 mode: "hybrid" (Huber + spectral_weight * AMSE excess) or "pure" 

109 (the AMSE formula alone). 

110 spectral_weight: weight of the spectral excess term in hybrid mode. 

111 delta: Huber transition point for the hybrid base term (matches 

112 the repo's default Huber loss). 

113 merge_bins_below: annuli with rounded |k| <= this value share one 

114 bin (the lowest annuli contain too few modes for a stable 

115 per-sample coherence estimate). 

116 eps: additive guard inside sqrt(P_k(x) * P_k(y)); keeps gradients 

117 finite when either field has an empty scale bin. 

118 static_ref_path: optional .npy path of a static reference field 

119 subtracted from both fields before the spectral terms. 

120 wavenumber_weight: per-bin upweighting of the spectral terms. 

121 None / "none" (default) leaves every annulus at weight 1, i.e. 

122 the unweighted Subich sum; "fastnet" applies 

123 gamma_k = max(N_k * k**sqrt(3), 1) (see ``_weight_bins``). 

124 

125 """ 

126 super().__init__() 

127 if mode not in ("hybrid", "pure"): 

128 msg = f"Unknown AMSELoss mode: {mode!r} (expected 'hybrid' or 'pure')" 

129 raise ValueError(msg) 

130 if merge_bins_below < 1: 

131 msg = f"merge_bins_below must be >= 1, got {merge_bins_below}" 

132 raise ValueError(msg) 

133 weight_mode = (wavenumber_weight or "none").lower() 

134 if weight_mode not in WAVENUMBER_WEIGHTS: 

135 msg = ( 

136 f"Unknown AMSELoss wavenumber_weight: {wavenumber_weight!r} " 

137 f"(expected None or one of {WAVENUMBER_WEIGHTS})" 

138 ) 

139 raise ValueError(msg) 

140 self.mode = mode 

141 self.spectral_weight = spectral_weight 

142 self.delta = delta 

143 self.merge_bins_below = merge_bins_below 

144 self.eps = eps 

145 self.wavenumber_weight = weight_mode 

146 # Optional anomaly transform: a static reference field (e.g. the 

147 # training-period mean SIC) subtracted from BOTH prediction and target 

148 # before the spectral terms. Removes the shared static structure 

149 # (coastline/climatological ring) that otherwise dominates band power 

150 # and inflates coherence, diluting the anti-blur excess (measured 

151 # 2026-07-27: static share 63-86% of band power on full_south). 

152 self.static_ref_path = static_ref_path 

153 self._static_ref: torch.Tensor | None = None 

154 # Lazily-built cache of binning tensors, keyed by (H, W, device). 

155 # A plain dict (not buffers) so it never enters the state dict. 

156 self._bin_cache: dict[tuple[int, int, str], BinCache] = {}

158 def _binning(self, height: int, width: int, device: torch.device) -> BinCache: 

159 """Return (bin index, mode weight, non-empty bin ids, gamma_k) for H x W.""" 

160 key = (height, width, str(device)) 

161 cached = self._bin_cache.get(key) 

162 if cached is not None: 

163 return cached 

164 ky = torch.fft.fftfreq(height, device=device) * height 

165 kx = torch.fft.rfftfreq(width, device=device) * width 

166 kmag = torch.sqrt(ky[:, None] ** 2 + kx[None, :] ** 2) 

167 bins = torch.round(kmag).long() 

168 # Merge the sparse low-|k| annuli (excluding DC) into one bin. 

169 bins = torch.where( 

170 (bins >= 1) & (bins < self.merge_bins_below), self.merge_bins_below, bins 

171 ) 

172 # rfft halves the x axis: interior columns stand for +-kx pairs and 

173 # count twice; column 0 and (for even W) the Nyquist column appear once. 

174 weights = torch.full(kmag.shape, 2.0, device=device) 

175 weights[:, 0] = 1.0 

176 if width % 2 == 0: 

177 weights[:, -1] = 1.0 

178 # The DC mode is excluded from the annuli (handled as a direct term). 

179 weights[0, 0] = 0.0 

180 bins_flat = bins.reshape(-1) 

181 weights_flat = weights.reshape(-1) 

182 counts = torch.zeros(int(bins_flat.max()) + 1, device=device) 

183 counts.scatter_add_(0, bins_flat, (weights_flat > 0).float()) 

184 present = torch.nonzero(counts > 0, as_tuple=False).squeeze(1) 

185 # FastNet wavenumber weight gamma_k = max(N_k * k**sqrt(3), 1), cached 

186 # alongside the bin tensors and used only when wavenumber_weight is on. 

187 # N_k is the number of modes of the FULL 2D DFT falling in bin k, so it 

188 # is the sum of the SAME rfft double-count weights used for the powers 

189 # (no second binning): interior columns count 2, column 0 and the even-W 

190 # Nyquist column count 1, and DC carries weight 0. k is the bin's 

191 # representative isotropic wavenumber, i.e. the bin index, which for the 

192 # merged low bin is merge_bins_below. 

193 mode_counts = torch.zeros_like(counts) 

194 mode_counts.scatter_add_(0, bins_flat, weights_flat) 

195 wavenumber = torch.arange(counts.numel(), device=device, dtype=counts.dtype) 

196 gamma = torch.clamp(mode_counts * wavenumber**FASTNET_EXPONENT, min=1.0) 

197 gamma = gamma.index_select(0, present) 

198 if self.wavenumber_weight != "none": 

199 # Once per grid, and only when the option is on: a run-log 

200 # fingerprint proving the flag actually reached the loss. 

201 logger.info( 

202 "AMSE wavenumber weighting %r engaged on %dx%d: %d active bins, " 

203 "gamma in [%.4g, %.4g]", 

204 self.wavenumber_weight, 

205 height, 

206 width, 

207 int(present.numel()), 

208 float(gamma.min()), 

209 float(gamma.max()), 

210 ) 

211 result = (bins_flat, weights_flat, present, gamma) 

212 self._bin_cache[key] = result 

213 return result

215 def _weight_bins( 

216 self, contributions: torch.Tensor, height: int, width: int 

217 ) -> torch.Tensor: 

218 """Apply the FastNet per-bin upweighting to [n_fields, n_bins] terms. 

219 

220 With ``wavenumber_weight="none"`` the input is returned untouched, so 

221 the default path is bit-for-bit the unweighted Subich sum. 

222 

223 With ``"fastnet"`` each bin is multiplied by 

224 ``gamma_k = max(N_k * k**sqrt(3), 1)`` and the result is divided by a 

225 DETACHED scalar chosen so the batch total is unchanged: 

226 

227 sum_(i,k) gamma_k * c_ik / Z == sum_(i,k) c_ik, 

228 Z = sum_(i,k) gamma_k * c_ik / sum_(i,k) c_ik. 

229 

230 Rationale: raw gamma_k grows like k**2.7 (N_k ~ 2*pi*k), spanning ~7 

231 orders of magnitude across the bins of a 432x432 grid, so the raw 

232 weighted sum would change the SCALE of the spectral term by orders of 

233 magnitude. That silently rescales the effective learning rate of the 

234 spectral branch and makes ``spectral_weight`` mean something different 

235 from its value in the unweighted arm - which would confound the 

236 factorial comparison the flag exists to serve. Normalising gamma to 

237 mean 1 over the active bins does NOT fix this: gamma is strongly 

238 anti-correlated with where the penalty sits (the penalty is concentrated 

239 at low |k| where gamma is smallest), so the total would still move by a 

240 large factor. Preserving the total makes the change purely a 

241 REDISTRIBUTION of the penalty across scales - exactly the intended 

242 intervention - and keeps the arm trainable at the unweighted arm's lr. 

243 Z is detached, so gradients are those of a fixed per-bin reweighting 

244 (a constant rescale of an ordinary weighted sum), not of a ratio. 

245 

246 The DC term is deliberately NOT weighted. It is not an annulus (the DC 

247 mode carries binning weight 0 and never reaches a bin), and the FastNet 

248 formula itself assigns it gamma_0 = max(0 * 0**sqrt(3), 1) = 1; it is 

249 the domain-mean bias, which must keep full weight. 

250 """ 

251 if self.wavenumber_weight == "none": 

252 return contributions 

253 gamma = self._binning(height, width, contributions.device)[3] 

254 weighted = contributions * gamma 

255 # Both sums are non-negative (every per-bin term is clamped at 0), so 

256 # the ratio is a weighted mean of gamma >= 1: bounded and well posed. 

257 # The eps pair makes Z = 1 for the exact-match case (0 / 0). 

258 scale = (weighted.detach().sum() + self.eps) / ( 

259 contributions.detach().sum() + self.eps 

260 ) 

261 return weighted / scale

263 def binned_spectra( 

264 self, prediction: torch.Tensor, target: torch.Tensor 

265 ) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]: 

266 """Per-field, per-bin spectral statistics of the mean-removed fields. 

267 

268 Returns: 

269 Tuple ``(P, T, C, dc)`` where ``P[i, b]`` / ``T[i, b]`` are the 

270 prediction/target powers of field ``i`` in |k|-bin ``b``, 

271 ``C[i, b]`` is the real cross-spectrum, and ``dc[i]`` is the 

272 squared difference of the field means. Powers are normalised so 

273 that ``sum_b (P + T - 2C) + dc`` equals the per-field MSE 

274 (Parseval's identity). 

275 

276 """ 

277 if prediction.shape != target.shape: 

278 msg = ( 

279 f"prediction shape {tuple(prediction.shape)} " 

280 f"!= target shape {tuple(target.shape)}" 

281 ) 

282 raise ValueError(msg) 

283 if prediction.ndim < MIN_FIELD_NDIM: 

284 msg = ( 

285 f"expected at least {MIN_FIELD_NDIM} dimensions, got {prediction.ndim}" 

286 ) 

287 raise ValueError(msg) 

288 height, width = prediction.shape[-2], prediction.shape[-1] 

289 fields_p = prediction.reshape(-1, height, width).float() 

290 fields_t = target.reshape(-1, height, width).float() 

291 if self.static_ref_path is not None: 

292 if self._static_ref is None: 

293 ref = torch.from_numpy(np.load(self.static_ref_path)).float() 

294 if not torch.isfinite(ref).all(): 

295 # A silently all-NaN reference field reached a production GPU job 

296 # on 2026-07-27 and NaN'd every loss while the model itself was 

297 # healthy. Fail loudly at load time instead. 

298 n_bad = int((~torch.isfinite(ref)).sum()) 

299 msg = ( 

300 f"static_ref at {self.static_ref_path} has {n_bad} non-finite " 

301 f"cells of {ref.numel()}; refusing to train on it." 

302 ) 

303 raise ValueError(msg) 

304 if ref.shape != (height, width): 

305 msg = ( 

306 f"static_ref shape {tuple(ref.shape)} does not match " 

307 f"field shape ({height}, {width})" 

308 ) 

309 raise ValueError(msg) 

310 self._static_ref = ref 

311 ref = self._static_ref.to(device=fields_p.device, dtype=fields_p.dtype) 

312 fields_p = fields_p - ref 

313 fields_t = fields_t - ref 

314 mean_p = fields_p.mean(dim=(-2, -1)) 

315 mean_t = fields_t.mean(dim=(-2, -1)) 

316 dc = (mean_p - mean_t) ** 2 

317 spec_p = torch.fft.rfft2(fields_p - mean_p[:, None, None], norm="ortho") 

318 spec_t = torch.fft.rfft2(fields_t - mean_t[:, None, None], norm="ortho") 

319 bins_flat, weights_flat, present, _ = self._binning( 

320 height, width, fields_p.device 

321 ) 

322 n_fields = spec_p.shape[0] 

323 n_bins_total = int(bins_flat.max()) + 1 

324 scale = 1.0 / (height * width) 

325 index = bins_flat.unsqueeze(0).expand(n_fields, -1) 

326 

327 def accumulate(values: torch.Tensor) -> torch.Tensor: 

328 out = torch.zeros(n_fields, n_bins_total, device=values.device) 

329 out.scatter_add_( 

330 1, index, (values * weights_flat).reshape(n_fields, -1) * scale 

331 ) 

332 return out.index_select(1, present) 

333 

334 # Avoid using .abs() and .conj() which are causing crashes on the DGX Spark 

335 # ("invalid value for --gpu-architecture") due to NVRTC JIT issues. 

336 power_p = accumulate((spec_p.real**2 + spec_p.imag**2).reshape(n_fields, -1)) 

337 power_t = accumulate((spec_t.real**2 + spec_t.imag**2).reshape(n_fields, -1)) 

338 cross = accumulate( 

339 (spec_p.real * spec_t.real + spec_p.imag * spec_t.imag).reshape( 

340 n_fields, -1 

341 ) 

342 ) 

343 return power_p, power_t, cross, dc

345 def spectral_excess( 

346 self, prediction: torch.Tensor, target: torch.Tensor 

347 ) -> torch.Tensor: 

348 """Per-field AMSE excess ``AMSE - MSE`` (the pure anti-blur surcharge). 

349 

350 Non-negative; zero iff, in every |k|-bin, the two power spectra match 

351 or the coherence is perfect. Shrinking predicted amplitude below the 

352 target's cannot reduce this term. 

353 """ 

354 power_p, power_t, cross, _ = self.binned_spectra(prediction, target) 

355 geo_mean = torch.sqrt(self.eps + power_p * power_t) 

356 coherence = torch.clamp(cross / geo_mean, max=1.0) 

357 excess_weight = (torch.maximum(power_p, power_t) - geo_mean).clamp_min(0.0) 

358 contributions = 2.0 * excess_weight * (1.0 - coherence) 

359 height, width = prediction.shape[-2], prediction.shape[-1] 

360 return self._weight_bins(contributions, height, width).sum(dim=1) 

361 

362 def pure_amse(self, prediction: torch.Tensor, target: torch.Tensor) -> torch.Tensor: 

363 """Per-field AMSE (amplitude + adjusted decorrelation terms + DC term).""" 

364 power_p, power_t, cross, dc = self.binned_spectra(prediction, target) 

365 geo_mean = torch.sqrt(self.eps + power_p * power_t) 

366 coherence = torch.clamp(cross / geo_mean, max=1.0) 

367 amplitude = (power_p + power_t - 2.0 * geo_mean).clamp_min(0.0) 

368 decorrelation = 2.0 * torch.maximum(power_p, power_t) * (1.0 - coherence) 

369 height, width = prediction.shape[-2], prediction.shape[-1] 

370 contributions = self._weight_bins(amplitude + decorrelation, height, width) 

371 return contributions.sum(dim=1) + dc

373 def forward(self, prediction: torch.Tensor, target: torch.Tensor) -> torch.Tensor: 

374 """Return the scalar loss for prediction/target of shape [..., H, W].""" 

375 if self.mode == "pure": 

376 return self.pure_amse(prediction, target).mean() 

377 base = functional.huber_loss( 

378 prediction.float(), target.float(), delta=self.delta 

379 ) 

380 return ( 

381 base 

382 + self.spectral_weight * self.spectral_excess(prediction, target).mean() 

383 )
