16@dataclass(frozen=True) 

17class CalendarDayClimatology: 

18 """Pixel-wise statistics for each calendar day, in the dataset's own units. 

19 

20 Dates are grouped by month/day label (``"MM-DD"``) rather than ordinal day-of-year, 

21 so 29 February keeps its own slot and every other calendar day stays aligned 

22 between leap and non-leap years. 

23 

24 Attributes: 

25 mean: [366, C, H, W] mean field for each calendar day. 

26 std: [366, C, H, W] population standard deviation for each calendar day. 

27 n_dates: [366] number of dates falling on each calendar day. 

28 

29 """ 

30 

31 # Map each month/day label to its index, enumerating the days of 2000 (a leap year) 

32 DAY_INDEX: ClassVar[dict[str, int]] = { 

33 str(day)[5:]: index 

34 for index, day in enumerate( 

35 np.arange( 

36 np.datetime64("2000-01-01"), 

37 np.datetime64("2001-01-01"), 

38 dtype="datetime64[D]", 

39 ) 

40 ) 

41 } 

42 # Triangular window half-width. Smooths day-to-day variability keeping seasonality. 

43 HALF_WINDOW: ClassVar[int] = 7 

44 

45 mean: ArrayTCHW 

46 std: ArrayTCHW 

47 n_dates: np.ndarray 

48 

49 @classmethod 

50 def day_index(cls, day: np.datetime64) -> int: 

51 """Return the 0-365 calendar-day index (month/day label) for a date.""" 

52 label = np.datetime_as_string(day.astype("datetime64[D]"), unit="D")[5:] 

53 return cls.DAY_INDEX[label] 

54 

55 @classmethod 

56 def from_dataset( 

57 cls, 

58 dataset: "SingleDataset", 

59 dates: Sequence[np.datetime64], 

60 ) -> Self: 

61 """Compute smoothed pixel-wise mean and standard deviation for each calendar day. 

62 

63 The data is read once, reducing it to a count, sum and sum of squares of the 

64 finite values on each calendar day. These are then smoothed around the 

65 366-day calendar with triangular weights, falling linearly from 1 at the centre 

66 to ``1 / (HALF_WINDOW + 1)`` at each end, so each calendar day draws on every 

67 date within ``HALF_WINDOW`` calendar days of it, wrapping from 31 December to 1 

68 January. Every year is treated as having a 29 February slot: that slot draws 

69 on 28 February and 1 March from every year, and in non-leap years those two 

70 days are treated as two days apart rather than one. 

71 

72 Sums are accumulated in float64. Non-finite values are excluded pixel by 

73 pixel, so a pixel is NaN only if it has no finite values within the window; 

74 the caller decides whether that is an error. 

75 

76 Args: 

77 dataset: The dataset to read fields from. 

78 dates: The dates to average over. 

79 

80 """ 

81 by_day: dict[int, list[np.datetime64]] = defaultdict(list) 

82 for day in dates: 

83 by_day[cls.day_index(day)].append(day) 

84 

85 # Reduce: count, sum and sum of squares of the finite values on each day 

86 counts, totals, squares = ( 

87 np.zeros((len(cls.DAY_INDEX), *dataset.space.chw), dtype=np.float64) 

88 for _ in range(3) 

89 ) 

90 n_dates = np.zeros(len(cls.DAY_INDEX), dtype=np.int64) 

91 for index, day_dates in by_day.items(): 

92 fields = dataset.get_tchw(day_dates).astype(np.float64) 

93 valid = np.isfinite(fields) 

94 values = np.where(valid, fields, 0.0) 

95 counts[index] = valid.sum(axis=0) 

96 totals[index] = values.sum(axis=0) 

97 squares[index] = (values**2).sum(axis=0) 

98 n_dates[index] = len(day_dates) 

99 

100 # Smooth: weighted sums over neighbouring calendar days, wrapping at year end 

101 def smooth(moment: np.ndarray) -> np.ndarray: 

102 smoothed = np.zeros_like(moment) 

103 for offset in range(-cls.HALF_WINDOW, cls.HALF_WINDOW + 1): 

104 weight = 1.0 - abs(offset) / (cls.HALF_WINDOW + 1) 

105 smoothed += weight * np.roll(moment, -offset, axis=0) 

106 return smoothed 

107 

108 weights = smooth(counts) 

109 # Pixels with no finite values in the window give 0/0, i.e. NaN 

110 with np.errstate(divide="ignore", invalid="ignore"): 

111 mean = smooth(totals) / weights 

112 variance = smooth(squares) / weights - mean**2 

113 

114 return cls( 

115 mean=mean.astype(np.float32), 

116 std=np.sqrt(np.maximum(variance, 0.0)).astype(np.float32), 

117 n_dates=n_dates, 

118 )
