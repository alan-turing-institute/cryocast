18class LatentObjective(TrainingObjective): 

19 """Use the loss that the processor calculates in latent space. 

20 

21 This objective is for processors that set `computes_loss_in_latent_space=True`. We 

22 encode the inputs over both the history and the forecast steps into latent space, 

23 and call `rollout()` on the processor, which returns both a latent prediction and a 

24 loss. We decode this prediction without gradients to calculate metrics. 

25 

26 The latent loss does not go through the decoder. Thus, we can add two auxiliary 

27 losses. We skip each one if its weight is zero. 

28 

29 - Reconstruction loss, weighted by `loss_weights.reconstruction`: we decode the 

30 encoded forecast-step inputs and compare them to the target. This loss trains the 

31 encoders and the decoder, and keeps the latent space compatible with the decoder. 

32 - Prediction loss, weighted by `loss_weights.prediction`: we decode the latent 

33 prediction from the processor, with gradients, and compare it to the target. Only 

34 the samples in `sample_mask` count. This loss trains the processor, the decoder 

35 and the encoders against the target in physical space. 

36 

37 If both weights are 0, nothing trains the decoder and the finaliser, so we freeze 

38 them. 

39 """ 

40 

41 # The latent loss compares forecasts against encoded inputs, so it needs inputs over 

42 # the forecast steps. 

43 needs_future_inputs = True 

44 

45 def __init__( 

46 self, 

47 model: "EncodeProcessDecode", 

48 loss_weights: Mapping[str, float] | None = None, 

49 ) -> None: 

50 """Validate the objective against the model, and freeze what it cannot train. 

51 

52 The objective does not keep a reference to the model. 

53 

54 Args: 

55 model: The model that owns this objective. 

56 loss_weights: Weights for the `reconstruction` and `prediction` loss terms 

57 (default 0.0 each). Weights must not be negative. 

58 

59 Raises: 

60 ValueError: If a key is unknown, a weight is negative, or the model uses 

61 physical rollout. 

62 

63 """ 

64 loss_weights = loss_weights or {} 

65 self.loss_weights = {"reconstruction": 0.0, "prediction": 0.0} 

66 for key, weight in loss_weights.items(): 

67 if key not in self.loss_weights: 

68 msg = ( 

69 f"Unknown loss_weights key '{key}'. Supported keys: " 

70 f"{sorted(self.loss_weights)}." 

71 ) 

72 raise ValueError(msg) 

73 if weight < 0: 

74 msg = f"loss_weights.{key} must not be negative, but it is {weight}." 

75 raise ValueError(msg) 

76 self.loss_weights[key] = float(weight) 

77 

78 # Physical rollout passes the output through the decoder, so the latent loss 

79 # cannot train it. 

80 if model.rollout_strategy.is_physical: 

81 msg = ( 

82 f"rollout_space='physical' is incompatible with processor " 

83 f"{type(model.processor).__name__}, which computes its loss in " 

84 f"latent space." 

85 ) 

86 raise ValueError(msg) 

87 

88 # Only the auxiliary losses use the decoder and the finaliser. If they are all 

89 # off, freeze both. 

90 if not any(self.loss_weights.values()): 

91 model.decoder.freeze() 

92 model.finaliser.freeze() 

93 

94 # Without the reconstruction loss, nothing keeps the latent space and the 

95 # decoder compatible while they train. 

96 if self.loss_weights["reconstruction"] == 0 and any( 

97 parameter.requires_grad 

98 for module in (*model.encoders, model.decoder) 

99 for parameter in module.parameters() 

100 ): 

101 log.warning( 

102 "loss_weights.reconstruction is 0, but the encoders or the " 

103 "decoder are trainable. Without the reconstruction loss, the latent " 

104 "space can drift away from what the decoder expects." 

105 ) 

106 

107 @override 

108 def training_step( 

109 self, 

110 model: "EncodeProcessDecode", 

111 batch: dict[str, TensorNTCHW], 

112 target: TensorNTCHW, 

113 ) -> tuple[TensorNTCHW, torch.Tensor]: 

114 # Encode history and future inputs into the combined latent space. 

115 history_latent = model._encode_inputs(batch) 

116 future_latent = self._encode_future_inputs( 

117 model, batch, with_grad=self.loss_weights["reconstruction"] > 0 

118 ) 

119 

120 # Process in latent space, detaching future inputs so that the latent loss 

121 # cannot decrease by changing its own target. 

122 processor_output = model.processor.rollout( 

123 history_latent, future_latent.detach() 

124 ) 

125 # Check that the processor returned a loss 

126 if (latent_loss := processor_output.loss) is None: 

127 msg = ( 

128 f"Processor {type(model.processor).__name__} promises to compute " 

129 "loss in latent space, but did not return one." 

130 ) 

131 raise ValueError(msg) 

132 # Check the number of forecast steps is in the expected range 

133 n_steps = processor_output.prediction.shape[1] 

134 if not 1 <= n_steps <= model.n_forecast_steps: 

135 msg = ( 

136 f"Processor {type(model.processor).__name__} returned a prediction " 

137 f"with {n_steps} forecast steps, but it must have between 1 and " 

138 f"{model.n_forecast_steps}." 

139 ) 

140 raise ValueError(msg) 

141 

142 # Use persistence as the skip-connection anchor, if required. 

143 anchor = model._anchor_ntchw(batch[model.output_space.name]) 

144 

145 # Relax the forecast towards climatology, if required. The reconstruction 

146 # loss decodes observations, not a forecast, so it is not relaxed. 

147 equilibrium = model._equilibrium(batch) 

148 

149 # Calculate each auxiliary loss that has a non-zero weight. 

150 losses: dict[str, torch.Tensor] = {} 

151 if self.loss_weights["reconstruction"] > 0: 

152 # Decode the encoded forecast-step inputs and compare them to the target. 

153 # With a PREVIOUS anchor, we use teacher forcing: each step after the first 

154 # uses the ground-truth for the previous forecast step, not the output of 

155 # the previous step. 

156 teacher_anchor = anchor 

157 if ( 

158 anchor is not None 

159 and model.rollout_strategy.anchor == SkipAnchorType.PREVIOUS 

160 ): 

161 teacher_anchor = torch.cat([anchor, target[:, :-1]], dim=1) 

162 reconstruction = model.finaliser.rollout( 

163 model.decoder.rollout(future_latent), anchor=teacher_anchor 

164 ) 

165 losses["reconstruction"] = model.loss(reconstruction, target) 

166 if self.loss_weights["prediction"] > 0: 

167 # Decode the differentiable prediction and compare it to the target 

168 losses["prediction"] = self._prediction_loss( 

169 model, 

170 processor_output.prediction, 

171 processor_output.sample_mask, 

172 target, 

173 anchor, 

174 equilibrium, 

175 ) 

176 

177 # Add the weighted auxiliary losses to the latent loss 

178 loss = sum( 

179 (self.loss_weights[name] * value for name, value in losses.items()), 

180 start=latent_loss, 

181 ) 

182 

183 # Log each term on its own if there are auxiliary losses. 

184 if losses: 

185 for name, value in {"latent": latent_loss, **losses}.items(): 

186 model.log_per_epoch(f"train_{name}_loss", value) 

187 

188 # Decode the prediction in evaluation mode and without gradients, so that 

189 # we do not affect BatchNorm running statistics in the decoder or finaliser. 

190 training_modes = { 

191 module: module.training for module in (model.decoder, model.finaliser) 

192 } 

193 for module in training_modes: 

194 module.eval() 

195 try: 

196 with torch.no_grad(): 

197 prediction = model._decode_forecast( 

198 processor_output.prediction, anchor, equilibrium 

199 ) 

200 finally: 

201 for module, training_mode in training_modes.items(): 

202 module.train(training_mode) 

203 # If the latent prediction has fewer forecast steps than the target, repeat 

204 # the last one until we have the correct number 

205 steps = torch.arange(model.n_forecast_steps, device=prediction.device) 

206 return prediction[:, steps.clamp(max=n_steps - 1)], loss

208 @staticmethod 

209 def _encode_future_inputs( 

210 model: "EncodeProcessDecode", 

211 batch: dict[str, TensorNTCHW], 

212 *, 

213 with_grad: bool = False, 

214 ) -> TensorNTCHW: 

215 """Encode every input over the forecast steps into the combined latent space. 

216 

217 This is the ground truth for the latent loss. 

218 

219 Args: 

220 model: The model that owns this objective. 

221 batch: Dictionary with a `future_<input>` TensorNTCHW entry per input 

222 dataset with shape (batch, n_forecast_steps, n_input_channels_k, H_input_k, W_input_k) 

223 with_grad: Whether to record gradients. Set this only when the result 

224 also goes into the reconstruction loss. The caller must then detach the 

225 copy that it gives to the processor as the latent target. 

226 

227 Returns: 

228 TensorNTCHW with shape (batch_size, n_forecast_steps, n_latent_channels_total, latent_height, latent_width) 

229 

230 """ 

231 try: 

232 future_inputs = { 

233 input_space.name: batch[f"future_{input_space.name}"] 

234 for input_space in model.input_spaces 

235 } 

236 except KeyError as exc: 

237 msg = ( 

238 f"Processor {type(model.processor).__name__} computes its loss in " 

239 f"latent space, so the batch needs future inputs, but {exc} is " 

240 f"missing." 

241 ) 

242 raise ValueError(msg) from exc 

243 

244 # Do not record gradients unless the reconstruction loss needs them. This saves 

245 # the memory for the encoder activations over the forecast steps. 

246 with torch.set_grad_enabled(with_grad and torch.is_grad_enabled()): 

247 return model._encode_inputs(future_inputs)

249 @staticmethod 

250 def _prediction_loss( 

251 model: "EncodeProcessDecode", 

252 prediction: TensorNTCHW, 

253 mask: torch.Tensor | None, 

254 target: TensorNTCHW, 

255 anchor: TensorNTCHW | None, 

256 equilibrium: TensorNTCHW | None, 

257 ) -> torch.Tensor: 

258 """Decode the processor's latent prediction and compare it to the target. 

259 

260 Only the samples in `mask` count. The prediction can hold fewer forecast steps 

261 than the target (for example, only the first step for an autoregressive 

262 processor), so we compare it to the same number of target steps. 

263 

264 Args: 

265 model: The model that owns this objective. 

266 prediction: Latent prediction from the processor training step, with 

267 gradients back to the processor. Shape (batch, T, C_latent, H, W), with 

268 T <= n_forecast_steps. 

269 mask: Boolean mask of shape (batch,), or None to use all samples. 

270 target: TensorNTCHW with shape (batch, n_forecast_steps, C_out, H, W). 

271 anchor: Skip-connection anchor with shape (batch, 1, C_out, H, W), or None. 

272 equilibrium: Climatology to relax towards, with shape (batch, 

273 n_forecast_steps, C_out, H, W), or None. 

274 

275 Returns: 

276 Scalar loss. It is 0 if no sample is in the mask. 

277 

278 Raises: 

279 ValueError: If the prediction has no gradients. 

280 

281 """ 

282 if not prediction.requires_grad: 

283 msg = ( 

284 f"loss_weights.prediction is greater than 0, but processor " 

285 f"{type(model.processor).__name__} returns a prediction without " 

286 f"gradients during training. Return a differentiable prediction or " 

287 f"set the weight to 0." 

288 ) 

289 raise ValueError(msg) 

290 

291 # The mean over an empty mask is NaN, so return 0. Keep the decoder and finaliser 

292 # parameters in the graph without decoding anything: with DDP, a parameter that 

293 # has no gradient on one process stops training. 

294 if mask is not None and not mask.any(): 

295 return ( 

296 sum( 

297 ( 

298 p.sum() 

299 for module in (model.decoder, model.finaliser) 

300 for p in module.parameters() 

301 if p.requires_grad 

302 ), 

303 start=prediction.new_zeros(()), 

304 ) 

305 * 0.0 

306 ) 

307 

308 # Decode only the samples in the mask. Without a mask, all samples count. 

309 samples = slice(None) if mask is None else mask 

310 decoded = model._decode_forecast( 

311 prediction[samples], 

312 None if anchor is None else anchor[samples], 

313 None if equilibrium is None else equilibrium[samples], 

314 ) 

315 return model.loss(decoded, target[samples, : prediction.shape[1]])
