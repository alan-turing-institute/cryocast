1"""Lead-time weighted wrapper around any loss. 

2 

3The wrapped base loss is evaluated independently at each forecast lead time 

4and the per-step values are combined with weights that grow with lead time: 

5 

6 w_t = (t + 1) ** exponent, t = 0, ..., T - 1 

7 

8rescaled to have mean 1. The weights are computed in log space (as a softmax over 

9``exponent * log(t + 1)``) so that large positive or negative exponents cannot overflow 

10or underflow to a NaN loss. The rescaling keeps the overall loss magnitude (and hence 

11the learning rate) comparable with the unwrapped loss: weighting only redistributes 

12emphasis between lead times. 

13 

14Wrapped losses that implement ``SupportsPerLeadTimeLoss`` are evaluated for every lead 

15time in one call; any other loss is called once per lead time. 

16""" 

17 

18import logging 

19import math 

20 

21import torch 

22from torch import nn 

23 

24from cryocast.types import NDIM_NTCHW, SupportsPerLeadTimeLoss 

25 

26log = logging.getLogger(__name__) 

27 

28 

29class LeadTimeWeightedLoss(nn.Module): 

30 """Weight a base loss function so that later lead times contribute more.""" 

31 

32 def __init__(self, wrapped_loss: nn.Module, exponent: float = 1.0) -> None: 

33 """Initialise the LeadTimeWeightedLoss. 

34 

35 Args: 

36 wrapped_loss: Loss function that takes an NCHW slice and returns a scalar. 

37 exponent: Power-law exponent for the weights. 0 gives uniform weighting, 

38 positive values emphasise later lead times, and negative values 

39 emphasise earlier lead times. Must be a finite number. 

40 

41 """ 

42 if not math.isfinite(exponent): 

43 msg = f"LeadTimeWeightedLoss exponent must be finite, but got {exponent}." 

44 raise ValueError(msg) 

45 # Weights have mean 1, so only a mean-reduced loss keeps the unweighted scale 

46 reduction = getattr(wrapped_loss, "reduction", "mean") 

47 if reduction == "none": 

48 msg = ( 

49 "LeadTimeWeightedLoss requires a wrapped loss that returns a scalar, " 

50 f"but {type(wrapped_loss).__name__} has reduction='none'." 

51 ) 

52 raise ValueError(msg) 

53 if reduction != "mean": 

54 msg = ( 

55 "LeadTimeWeightedLoss works best with a mean-reduced wrapped loss, but " 

56 f"{type(wrapped_loss).__name__} has reduction={reduction!r}. This may " 

57 "cause the overall scale to change with the number of forecast steps." 

58 ) 

59 log.warning(msg) 

60 super().__init__() 

61 self._exponent = exponent 

62 self._weights_cache: dict[tuple[int, torch.device], torch.Tensor] = {} 

63 self._wrapped_loss = wrapped_loss 

64 

65 @property 

66 def exponent(self) -> float: 

67 return self._exponent 

68 

69 def forward(self, prediction: torch.Tensor, target: torch.Tensor) -> torch.Tensor: 

70 """Return the lead-time weighted scalar loss for NTCHW prediction/target.""" 

71 if prediction.ndim != NDIM_NTCHW: 

72 msg = ( 

73 "LeadTimeWeightedLoss expects NTCHW inputs with a lead-time dimension, " 

74 f"but got a tensor of shape {tuple(prediction.shape)}." 

75 ) 

76 raise ValueError(msg) 

77 if prediction.shape != target.shape: 

78 msg = ( 

79 "LeadTimeWeightedLoss expects prediction and target to have the same " 

80 f"shape, but got {tuple(prediction.shape)} and {tuple(target.shape)}." 

81 ) 

82 raise ValueError(msg) 

83 # Derive the number of steps from the input for increased flexibility 

84 n_steps = prediction.shape[1] 

85 if isinstance(self._wrapped_loss, SupportsPerLeadTimeLoss): 

86 per_step = self._wrapped_loss.per_lead_time_loss(prediction, target) 

87 else: 

88 per_step = torch.stack( 

89 [ 

90 self._wrapped_loss(prediction[:, t], target[:, t]) 

91 for t in range(n_steps) 

92 ] 

93 ) 

94 return (self.weights(n_steps, per_step.device) * per_step).mean() 

95 

96 def weights(self, n_steps: int, device: torch.device) -> torch.Tensor: 

97 """Return the per-lead-time weights for `n_steps` forecast steps.""" 

98 cache_key = (n_steps, device) 

99 if cache_key not in self._weights_cache: 

100 # Disable inference mode so weights can be used in validation and training 

101 with torch.inference_mode(mode=False): 

102 # Numerically stable equivalent to steps**exponent / sum(steps**exponent) 

103 log_steps = torch.arange( 

104 1, n_steps + 1, device=device, dtype=torch.float32 

105 ).log() 

106 self._weights_cache[cache_key] = ( 

107 torch.softmax(self.exponent * log_steps, dim=0) * n_steps 

108 ) 

109 return self._weights_cache[cache_key]
