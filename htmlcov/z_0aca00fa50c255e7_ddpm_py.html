1"""Latent-space DDPM processor. 

2 

3This module provides a diffusion-based processor for the EncodeProcessDecode 

4pipeline. It performs denoising in the shared latent space produced by the 

5encoders, using v-parameterisation with `UNetDiffusion` and 

6`GaussianDiffusion`. 

7 

8The processor supports two forecasting modes: 

9 

10* Autoregressive: generates forecast steps sequentially, updating the history 

11 with each predicted target latent. 

12* Parallel: generates all forecast steps jointly in a single diffusion 

13 process, with forecast steps folded into the channel dimension. 

14 

15Only the target-latent channels are generated by the diffusion model. 

16Non-target channels are not predicted; they are copied from the last history 

17frame and carried forward unchanged. 

18""" 

19

33class DDPMProcessor(BaseProcessor): 

34 """Latent-space DDPM processor with v-prediction. 

35 

36 Input space: 

37 TensorNTCHW with shape (batch_size, n_history_steps, n_latent_channels_total, latent_height, latent_width) 

38 - Concatenation of every encoder's latent output along the channel dimension 

39 - The target-latent slice is assumed to occupy channels [target_slice_start : target_slice_start + n_latent_channels_target] 

40 

41 Output space: 

42 TensorNTCHW with shape (batch_size, n_forecast_steps, n_latent_channels_total, latent_height, latent_width) 

43 - Target-latent slice contains the denoised prediction from reverse diffusion 

44 - Non-target channels are carried forward from the last observed history frame (persistence in latent) 

45 """

47 def __init__( # noqa: PLR0913 

48 self, 

49 *, 

50 timesteps: int = 1000, 

51 beta_schedule: str = "cosine", 

52 kernel_size: int = 3, 

53 start_out_channels: int = 64, 

54 time_embed_dim: int = 256, 

55 dropout_rate: float = 0.1, 

56 normalization: str = "groupnorm", 

57 activation: str = "SiLU", 

58 use_autoregressive: bool = True, 

59 loss: DictConfig | nn.Module, 

60 **kwargs: Any, 

61 ) -> None: 

62 """Initialize the DDPM processor. 

63 

64 Args: 

65 timesteps (int): Number of diffusion timesteps. Default is 1000. 

66 beta_schedule (str): Beta schedule used by ``GaussianDiffusion``: 

67 ``"cosine"`` or ``"linear"``. Default is ``"cosine"``. 

68 kernel_size (int): Convolution kernel size used in the conditional UNet. 

69 start_out_channels (int): Base number of channels in the first UNet block. 

70 time_embed_dim (int): Dimensionality of the diffusion timestep embedding. 

71 dropout_rate (float): Dropout probability applied inside the UNet blocks. 

72 normalization (str): Normalization layer type (e.g., "groupnorm"). 

73 activation (str): Activation function used throughout the network (e.g., "SiLU"). 

74 use_autoregressive (bool): Whether to use autoregressive sampling and 

75 one-step training. Default is True. 

76 loss (DictConfig | nn.Module): Loss module applied to (pred_v, 

77 target_v). Set this to ``${loss}`` in the yaml to reuse the 

78 top-level configured loss. 

79 **kwargs: Additional arguments passed to ``BaseProcessor``. 

80 

81 """ 

82 kwargs.pop("computes_loss_in_latent_space", None) 

83 super().__init__(computes_loss_in_latent_space=True, **kwargs) 

84 

85 # Instantiate the configured loss function. 

86 self.loss_fn: nn.Module = ( 

87 loss if isinstance(loss, nn.Module) else hydra.utils.instantiate(loss) 

88 ) 

89 

90 # c_combined: total channels across all encoder latents concatenated. 

91 c_combined = self.data_space.channels 

92 

93 # c_target: channels of the target encoder's latent output. 

94 c_target = self.data_space_target.channels 

95 

96 # Validate that the target slice fits inside the combined latent channels. 

97 if ( 

98 self.target_channel_offset is None 

99 or self.target_channel_offset < 0 

100 or self.target_channel_offset + c_target > c_combined 

101 ): 

102 msg = ( 

103 f"target_channel_offset={self.target_channel_offset} with target channels=" 

104 f"{c_target} does not fit inside combined latent channels=" 

105 f"{c_combined}." 

106 ) 

107 raise ValueError(msg) 

108 

109 # Save the position of the target slice in the combined latent space 

110 self.target_slice_start = self.target_channel_offset 

111 self.target_slice_end = self.target_channel_offset + c_target 

112 self.c_target = c_target 

113 self.c_combined = c_combined 

114 

115 self.timesteps = timesteps 

116 self.use_autoregressive = use_autoregressive 

117 

118 # UNet conditioning channels: history folded NTCHW -> NCHW. 

119 cond_channels = c_combined * self.n_history_steps 

120 

121 # Channels the UNet denoises: c_target (AR) or n_forecast_steps * c_target (parallel, all steps folded into channels). 

122 diffused_channels = ( 

123 c_target if use_autoregressive else c_target * self.n_forecast_steps 

124 ) 

125 self.diffused_channels = diffused_channels 

126 

127 self.model = UNetDiffusion( 

128 input_channels=cond_channels, 

129 output_channels=diffused_channels, 

130 timesteps=timesteps, 

131 kernel_size=kernel_size, 

132 start_out_channels=start_out_channels, 

133 time_embed_dim=time_embed_dim, 

134 normalization=normalization, 

135 activation=activation, 

136 dropout_rate=dropout_rate, 

137 ) 

138 self.diffusion = GaussianDiffusion( 

139 timesteps=timesteps, 

140 beta_schedule=BetaSchedule(beta_schedule), 

141 )

143 def _build_combined_latent( 

144 self, target_ntchw: TensorNTCHW, last_frame: TensorNCHW 

145 ) -> TensorNTCHW: 

146 """Lift target-latent forecasts into the combined latent representation. 

147 

148 Non-target channels are carried forward from ``last_frame`` (persistence 

149 in latent space) for every forecast step, while the target-latent slice 

150 is overwritten with ``target_ntchw``. 

151 

152 Args: 

153 target_ntchw (TensorNTCHW): Target-latent forecasts of shape 

154 (B, n_forecast_steps, C_target, H, W). 

155 last_frame (TensorNCHW): Last observed history frame of shape 

156 (B, C_combined, H, W). 

157 

158 Returns: 

159 TensorNTCHW: Combined-latent tensor of shape 

160 (B, n_forecast_steps, C_combined, H, W). 

161 

162 """ 

163 b, _, h, w = last_frame.shape 

164 expanded: TensorNTCHW = last_frame.unsqueeze(1).expand( 

165 b, self.n_forecast_steps, self.c_combined, h, w 

166 ) 

167 return self._insert_target(expanded, target_ntchw)

169 def _build_metrics_prediction( 

170 self, pred_x0: TensorNCHW, last_frame: TensorNCHW 

171 ) -> TensorNTCHW: 

172 """Lift the training-time x_0 estimate into an NTCHW combined-latent tensor. 

173 

174 The returned tensor is only used for metrics/callbacks (the actual 

175 training signal is the v-prediction MSE loss). Non-target channels are 

176 filled with the last observed frame (persistence in latent space). 

177 

178 Args: 

179 pred_x0 (TensorNCHW): Predicted x_0 of shape (B, C_target, H, W) 

180 in autoregressive mode, or (B, n_forecast_steps * C_target, H, W) 

181 in parallel mode. 

182 last_frame (TensorNCHW): Last observed history frame of shape 

183 (B, C_combined, H, W). 

184 

185 Returns: 

186 TensorNTCHW: Combined-latent tensor of shape 

187 (B, n_forecast_steps, C_combined, H, W). 

188 

189 """ 

190 b = pred_x0.shape[0] 

191 h, w = last_frame.shape[-2:] 

192 

193 # Restore the predicted target latent to the combined latent representation. 

194 if self.use_autoregressive: 

195 # In autoregressive mode, we duplicate predicted x_0 across all forecast steps 

196 pred_x0_ntchw = pred_x0.unsqueeze(1).expand( 

197 b, self.n_forecast_steps, self.c_target, h, w 

198 ) 

199 else: 

200 # In parallel mode, the predicted x_0 is folded into channels for all forecast steps. 

201 pred_x0_ntchw = pred_x0.reshape( 

202 b, self.n_forecast_steps, self.c_target, h, w 

203 ) 

204 return self._build_combined_latent(pred_x0_ntchw, last_frame)

206 def _insert_target( 

207 self, base_frame: TensorNCHW | TensorNTCHW, target: TensorNCHW | TensorNTCHW 

208 ) -> TensorNCHW | TensorNTCHW: 

209 """Return a copy of ``base_frame`` with its target slice overwritten by ``target``. 

210 

211 Used to produce a C_combined-channel prediction for the frozen decoder while 

212 only forecasting the target-latent slice. Since the target-channel dimension is 

213 always third-from-last, so this works for both a single timestep (NCHW) and a 

214 series of timestep steps (NTCHW). 

215 

216 Args: 

217 base_frame: Combined-latent tensor of shape (..., C_combined, H, W). 

218 target: Target-latent slice of shape (..., C_target, H, W). 

219 

220 Returns: 

221 torch.Tensor: Combined-latent tensor of shape (..., C_combined, H, W). 

222 

223 """ 

224 result = base_frame.clone() 

225 result[..., self.target_slice_start : self.target_slice_end, :, :] = target 

226 return result

228 def _rollout_inference(self, x: TensorNTCHW) -> ProcessorOutput: 

229 """Generate a latent forecast using a reverse diffusion process. 

230 

231 This method selects between two diffusion sampling strategies: 

232 

233 1. Non-autoregressive (parallel) sampling: 

234 - The model generates the entire future sequence in a single diffusion process. 

235 - No temporal dependency exists between forecast steps. 

236 

237 2. Autoregressive sampling: 

238 - Forecast steps are generated sequentially. 

239 - Each step is produced via an independent diffusion process. 

240 - The conditioning window is updated after each step to incorporate 

241 previously generated outputs. 

242 

243 Args: 

244 x (TensorNTCHW): Encoded history tensor of shape 

245 (B, n_history_steps, n_latent_channels_total, H, W). 

246 

247 Returns: 

248 ProcessorOutput: 

249 - prediction: (B, n_forecast_steps, n_latent_channels_total, H, W) 

250 - loss: None 

251 

252 Notes: 

253 - The diffusion process follows v-parameterization. 

254 - Sampling begins from standard Gaussian noise. 

255 

256 """ 

257 if self.use_autoregressive: 

258 return ProcessorOutput(prediction=self._sample_autoregressive(x)) 

259 return ProcessorOutput(prediction=self._sample_parallel(x))

261 def _rollout_training(self, x: TensorNTCHW, y: TensorNTCHW) -> ProcessorOutput: 

262 """One training rollout using DDPM v-prediction loss. 

263 

264 During training, the clean target latent is corrupted using the forward 

265 diffusion process by adding noise at a randomly sampled timestep. The 

266 model is trained to predict the corresponding v-target on the 

267 target-latent slice. 

268 

269 Args: 

270 x (TensorNTCHW): Encoded history tensor of shape 

271 (B, n_history_steps, n_latent_channels_total, H, W). 

272 y (TensorNTCHW): Encoded target latent of shape 

273 (B, n_forecast_steps, n_latent_channels_target, H, W). 

274 

275 Returns: 

276 ProcessorOutput: 

277 - prediction: (B, n_forecast_steps, n_latent_channels_total, H, W) 

278 Combined-latent tensor built from the predicted x_0 (target 

279 slice) and the last observed frame (non-target slice). Used 

280 only for metrics/callbacks under ``torch.no_grad``. 

281 - loss: v-prediction loss value computed by ``self.loss_fn``. 

282 

283 Notes: 

284 - In autoregressive mode, the model is trained on forecast step 0 

285 only, matching the convention used by ``icenet_mp.models.ddpm.DDPM``. 

286 Multi-step forecasts come from rolling out the AR loop at inference. 

287 - In parallel mode, the model is trained on all forecast steps 

288 jointly (folded into the channel dimension). 

289 

290 """ 

291 b = x.shape[0] 

292 device = x.device 

293 

294 # Last observed frame; its non-target channels are copied into the metrics prediction 

295 # (the model only predicts the target slice, so the rest is persistence). 

296 last_frame = x[:, -1] # (B, C_combined, H, W) 

297 

298 # History frames folded into channels for the 2D UNet. 

299 cond = x.flatten(start_dim=1, end_dim=2) # (B, T_hist * C_combined, H, W) 

300 

301 # Clean target to denoise: AR uses step 0 only; parallel folds all steps into channels. 

302 if self.use_autoregressive: 

303 # AR trains on forecast step 0 (T=0) only. 

304 y_flat = y[:, 0] # (B, C_target, H, W) 

305 else: 

306 y_flat = y.reshape(b, self.n_forecast_steps * self.c_target, *y.shape[-2:]) 

307 

308 # Random diffusion timestep per sample. 

309 t = torch.randint(0, self.timesteps, (b,), device=device).long() 

310 

311 # Add noise to the clean target at diffusion step t. 

312 noise = torch.randn_like(y_flat) 

313 noisy_y = self.diffusion.q_sample(y_flat, t, noise) 

314 

315 # Predict the velocity conditioned on the noisy target and history. 

316 pred_v = self.model(noisy_y, t, cond) 

317 

318 # Compute the target velocity for the sampled noise and timestep. 

319 target_v = self.diffusion.calculate_v(y_flat, noise, t) 

320 

321 # Compute the v-prediction training loss. 

322 loss = self.loss_fn(pred_v, target_v) 

323 

324 # Reconstruct x0 for metrics only; this is not used for training. 

325 with torch.no_grad(): 

326 # calculate_v() has the same formula as the x_0 reconstruction, 

327 # so we reuse it by passing pred_v as x_start. 

328 pred_x0 = self.diffusion.calculate_v(x_start=pred_v, noise=noisy_y, t=t) 

329 prediction = self._build_metrics_prediction(pred_x0, last_frame) 

330 

331 return ProcessorOutput(prediction=prediction, loss=loss)

333 def _run_reverse_diffusion(self, y: TensorNCHW, cond: TensorNCHW) -> TensorNCHW: 

334 """Iteratively denoise ``y`` from t=timesteps-1 down to t=0. 

335 

336 Args: 

337 y (TensorNCHW): Noisy latent to denoise, of shape (B, C, H, W). 

338 cond (TensorNCHW): History condition folded to channels, of shape 

339 (B, T_hist * C_combined, H, W). 

340 

341 Returns: 

342 TensorNCHW: Denoised latent of shape (B, C, H, W). 

343 

344 """ 

345 b = y.shape[0] 

346 device = y.device 

347 for t_step in reversed(range(self.timesteps)): 

348 t = torch.full((b,), t_step, dtype=torch.long, device=device) 

349 pred_v = self.model(y, t, cond) 

350 y = self.diffusion.p_sample(y, t, pred_v) 

351 return y

353 def _sample_autoregressive(self, x: TensorNTCHW) -> TensorNTCHW: 

354 """Autoregressive reverse diffusion sampling (one forecast step at a time). 

355 

356 Each forecast step is generated sequentially via an independent reverse 

357 diffusion process. After each step, the conditioning window is updated: 

358 the predicted target-latent frame is appended to the history's target 

359 slice, and non-target channels are carried forward from the last 

360 observed frame (persistence in latent space). 

361 

362 Args: 

363 x (TensorNTCHW): Encoded history tensor of shape 

364 (B, n_history_steps, n_latent_channels_total, H, W). 

365 

366 Returns: 

367 TensorNTCHW: Denoised forecast latent of shape 

368 (B, n_forecast_steps, n_latent_channels_total, H, W), 

369 formed by stacking all per-step predictions along the time 

370 dimension and lifting into the combined latent. 

371 

372 Notes: 

373 - Sampling at each step starts from standard Gaussian noise. 

374 - The model predicts v-parameterization at every diffusion timestep. 

375 - The history window slides forward by one frame per step, using 

376 the model's own prediction as the new observation of the target 

377 slice. 

378 

379 """ 

380 cond_window: TensorNTCHW = x.clone() # (B, T_hist, C_combined, H, W) 

381 b, _, _, h, w = x.shape 

382 device = x.device 

383 all_predictions: list[TensorNCHW] = [] 

384 

385 for _ in range(self.n_forecast_steps): 

386 cond = cond_window.flatten( 

387 start_dim=1, end_dim=2 

388 ) # (B, T_hist * C_combined, H, W) 

389 

390 # Start each forecast step from Gaussian noise. 

391 y = torch.randn((b, self.c_target, h, w), device=device) 

392 

393 # Iteratively denoise from the final diffusion timestep to zero. 

394 y = self._run_reverse_diffusion(y, cond) 

395 

396 all_predictions.append(y) 

397 

398 # Slide the history window forward: drop the oldest frame and 

399 # append the new prediction for the next forecast step. 

400 last_frame: TensorNCHW = cond_window[:, -1] # (B, C_combined, H, W) 

401 new_frame: TensorNCHW = self._insert_target(last_frame, y).unsqueeze(1) 

402 cond_window = torch.cat([cond_window[:, 1:], new_frame], dim=1) 

403 

404 target_ntchw = torch.stack( 

405 all_predictions, dim=1 

406 ) # (B, T_fcst, C_target, H, W) 

407 return self._build_combined_latent(target_ntchw, x[:, -1])

409 def _sample_parallel(self, x: TensorNTCHW) -> TensorNTCHW: 

410 """Non-autoregressive (parallel) reverse diffusion sampling. 

411 

412 This method generates the entire forecast sequence in a single 

413 diffusion process applied to one joint output tensor. The forecast 

414 steps are encoded as channels in a single tensor of shape 

415 (B, n_forecast_steps * C_target, H, W), then reshaped back to NTCHW 

416 and lifted into the combined latent for the frozen decoder. 

417 

418 Args: 

419 x (TensorNTCHW): Encoded history tensor of shape 

420 (B, n_history_steps, n_latent_channels_total, H, W). 

421 

422 Returns: 

423 TensorNTCHW: Denoised forecast latent of shape 

424 (B, n_forecast_steps, n_latent_channels_total, H, W). 

425 

426 Notes: 

427 - Sampling starts from Gaussian noise. 

428 - The model predicts v-parameterization at each diffusion step. 

429 - All forecast steps are denoised together as a single object. 

430 

431 """ 

432 cond = x.flatten(start_dim=1, end_dim=2) 

433 

434 b, _, h, w = cond.shape 

435 device = cond.device 

436 

437 y = torch.randn((b, self.n_forecast_steps * self.c_target, h, w), device=device) 

438 

439 # Iteratively denoise from the final diffusion timestep to zero. 

440 y = self._run_reverse_diffusion(y, cond) 

441 

442 # Build the combined latent, keeping non-target channels persistent. 

443 y_ntchw = y.reshape(b, self.n_forecast_steps, self.c_target, h, w) 

444 return self._build_combined_latent(y_ntchw, x[:, -1])

446 def rollout( 

447 self, 

448 x: TensorNTCHW, 

449 y: TensorNTCHW | None = None, 

450 ) -> ProcessorOutput: 

451 """Run the diffusion process in latent space. 

452 

453 Dispatches between two execution paths: 

454 

455 1. Training path (``y`` provided): 

456 - Adds noise to the encoded target latent at a random diffusion timestep. 

457 - Predicts v conditioned on the encoded history. 

458 - Returns a v-prediction loss inside ``ProcessorOutput``, computed 

459 by ``self.loss_fn`` (the configured loss). 

460 

461 2. Inference path (``y`` is None): 

462 - Runs reverse diffusion from Gaussian noise conditioned on the encoded history. 

463 - Autoregressive or parallel depending on ``self.use_autoregressive``. 

464 

465 Args: 

466 x (TensorNTCHW): Encoded history tensor of shape 

467 (B, n_history_steps, n_latent_channels_total, H, W). 

468 y (TensorNTCHW | None): During training: encoded target latent of 

469 shape (B, n_forecast_steps, n_latent_channels_target, H, W). 

470 Otherwise: None. 

471 

472 Returns: 

473 ProcessorOutput: 

474 - prediction: (B, n_forecast_steps, n_latent_channels_total, H, W) 

475 - loss: v-prediction loss from ``self.loss_fn`` (training only, 

476 otherwise None) 

477 

478 """ 

479 if y is not None: 

480 return self._rollout_training(x, y) 

481 return self._rollout_inference(x)
