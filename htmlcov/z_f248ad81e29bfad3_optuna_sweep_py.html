95 @cached_property 

96 def _built_parameters(self) -> list[Parameter]: 

97 """Build and validate every configured Parameter. 

98 

99 Raises if two parameters sanitise to the same W&B config key, since the sweep 

100 or trial config would otherwise silently drop one of them. 

101 """ 

102 parameters = [ 

103 build_parameter(name, param_spec) 

104 for name, param_spec in self.parameters.items() 

105 ] 

106 seen: dict[str, str] = {} 

107 for parameter in parameters: 

108 collision = seen.get(parameter.sanitised_name) 

109 if collision is not None: 

110 msg = ( 

111 f"Parameters '{collision}' and '{parameter.name}' both sanitise " 

112 f"to the W&B config key '{parameter.sanitised_name}'. If no " 

113 "changes are made, one will be silently dropped from the sweep." 

114 ) 

115 raise ValueError(msg) 

116 seen[parameter.sanitised_name] = parameter.name 

117 return parameters

119 def ask(self) -> tuple[Trial, list[tuple[Parameter, int | float | str]]]: 

120 """Ask the study for a new trial and suggest a value for each parameter. 

121 

122 ask() and suggest() calls both mutate the sampler, so we use a single lock 

123 acquisition to keep the pair synchronised. 

124 """ 

125 with self.sampler.lock(self.study): 

126 trial = self.study.ask() 

127 overrides = [ 

128 (parameter, parameter.suggest(trial)) 

129 for parameter in self._built_parameters 

130 ] 

131 return trial, overrides

133 def generate_trial_config( 

134 self, overrides: list[tuple[Parameter, int | float | str]] 

135 ) -> DictConfig: 

136 """Generate a trial config from a list of parameter overrides.""" 

137 base_config = DictConfig(OmegaConf.load(self.study_path / "model_config.yaml")) 

138 # Set each override directly, so that categorical values cannot be interpreted 

139 # as YAML keywords (e.g. "off", "null") and re-parsed into a bool/None. We set 

140 # struct-mode so that typos will raise instead of silently creating new keys. 

141 OmegaConf.set_struct(base_config, value=True) 

142 for parameter, value in overrides: 

143 OmegaConf.update(base_config, parameter.name, value, merge=True) 

144 # Disable struct mode before merging in wandb config, which is a new key 

145 OmegaConf.set_struct(base_config, value=False) 

146 wandb_overrides = OmegaConf.create( 

147 { 

148 "reporting": { 

149 "loggers": { 

150 "wandb": { 

151 "config": { 

152 parameter.sanitised_name: value 

153 for parameter, value in overrides 

154 } 

155 } 

156 } 

157 } 

158 } 

159 ) 

160 return DictConfig(OmegaConf.merge(base_config, wandb_overrides))

162 def validate_parameters(self, model_cfg: DictConfig) -> None: 

163 """Check that every configured parameter resolves against `model_cfg`.""" 

164 probe_config = model_cfg.copy() 

165 OmegaConf.set_struct(probe_config, value=True) 

166 errors: list[str] = [] 

167 for parameter in self._built_parameters: 

168 try: 

169 OmegaConf.update( 

170 probe_config, 

171 parameter.name, 

172 parameter.probe_value(), 

173 merge=True, 

174 ) 

175 except OmegaConfBaseException as exc: 

176 errors.append(f"'{parameter.name}': {exc}") 

177 if errors: 

178 msg = "Invalid sweep parameter(s):\n" + "\n".join(errors) 

179 raise ValueError(msg)

205 def initialise_sweep(self, model_cfg: DictConfig) -> str: 

206 """Generate a new W&B sweep.""" 

207 # Generate the W&B sweep config 

208 sweep_config = { 

209 "program": "imp", 

210 "method": "random", 

211 "metric": self.metric, 

212 "parameters": { 

213 k: v 

214 for parameter in self._built_parameters 

215 for k, v in parameter.sweep_cfg().items() 

216 }, 

217 } 

218 wandb_cfg = model_cfg.get("reporting", {}).get("loggers", {}).get("wandb", {}) 

219 self._entity = wandb_cfg.get("entity") or os.environ.get("WANDB_ENTITY") 

220 # Start the sweep 

221 return wandb.sweep(sweep_config, entity=self.entity, project="train")
